function parseCSV(text) {
const rows = [];
let row = [], field = "", inQuotes = false;
for (let i = 0; i < text.length; i++) {
const c = text[i];
if (inQuotes) {
if (c === '"') {
if (text[i + 1] === '"') { field += '"'; i++; }
else inQuotes = false;
} else field += c;
continue;
}
if (c === '"') { inQuotes = true; continue; }
if (c === ',') { row.push(field); field = ""; continue; }
if (c === '\r') continue;
if (c === '\n') { row.push(field); rows.push(row); row = []; field = ""; continue; }
field += c;
}
if (field.length || row.length) { row.push(field); rows.push(row); }
return rows;
}
/* Columns located by header name, never by fixed position. */
function indexHeaders(headerRow) {
const map = {};
headerRow.forEach((h, i) => { map[String(h).trim().toUpperCase()] = i; });
return map;
}
function cell(row, map, name) {
const i = map[name];
return i === undefined ? "" : String(row[i] ?? "").trim();
}
function toNumber(v) {
const digits = String(v ?? "").replace(/[^\d.-]/g, "");
const n = Number(digits);
return isFinite(n) ? n : 0;
}
const euro = new Intl.NumberFormat("en-IE", { maximumFractionDigits: 0 });
function formatEuro(v) { return "\u20AC" + euro.format(toNumber(v)); }
/* Exact equality only: "NOT LIVE" contains "LIVE". */
function isTrue(v) { return String(v ?? "").trim().toUpperCase() === "TRUE"; }
function isOn(v) { return String(v ?? "").trim().toUpperCase() === "ON"; }
function escapeHtml(s) {
return String(s ?? "")
.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;")
.replace(/"/g, "&quot;").replace(/'/g, "&#39;");
}
/* ---------- fetch ---------- */
async function fetchItems() {
const res = await fetch(URL_BROADCAST + "&cb=" + Date.now(), { cache: "no-store" });
if (!res.ok) throw new Error("items " + res.status);
const rows = parseCSV(await res.text());
if (rows.length < 2) throw new Error("items empty");
const map = indexHeaders(rows[0]);
const out = [];
for (let i = 1; i < rows.length; i++) {
const r = rows[i];
const num = cell(r, map, "ITEM NUM");
if (!num) continue;
out.push({
num: num,
name: cell(r, map, "THIS COLUMN IS THE MASTER DESCRIPTION"),
video: cell(r, map, "VIDEO_URL"),
sponsor: cell(r, map, "SPONSOR NAME"),
logo: cell(r, map, "SPONSOR LOGO URL"),
bullets: cell(r, map, "ITEM DESCRIPTION BULLETS"),
live: isTrue(cell(r, map, "LIVE AUCTION")),
sold: isTrue(cell(r, map, "SOLD")),
bid: cell(r, map, "BID FOR DISPLAY")
});
}
if (!out.length) throw new Error("no item rows");
return out;
}
async function fetchControl() {
const res = await fetch(URL_CONTROL + "&cb=" + Date.now(), { cache: "no-store" });
if (!res.ok) throw new Error("control " + res.status);
const rows = parseCSV(await res.text());
const obj = {};
rows.forEach(r => {
const k = String(r[0] ?? "").trim().toUpperCase();
if (k) obj[k] = String(r[1] ?? "").trim();
});
return obj;
}
async function refresh() {
try {
const [newItems, newControl] = await Promise.all([fetchItems(), fetchControl()]);
const firstLoad = items.length === 0;
items = newItems;
control = newControl;
lastGoodFetch = Date.now();
applyControl();
if (firstLoad) startRotation();
} catch (e) {
/* Keep last known data on screen. Never blank the display. */
console.error("fetch failed:", e.message);
}
updateStaleFlag();
}
function updateStaleFlag() {
const stale = lastGoodFetch && (Date.now() - lastGoodFetch > STALE_AFTER_MS);
document.getElementById("stale-flag").classList.toggle("is-visible", !!stale);
}
/* ---------- control signals ---------- */
function applyControl() {
const closing = isOn(control["CLOSING_SOON"]);
const closed = isOn(control["AUCTION_CLOSED"]);
const cb = document.getElementById("banner-closing");
cb.classList.toggle("is-visible", closing && !closed);
cb.classList.toggle("banner-pulse", closing && !closed);
document.getElementById("banner-closed").classList.toggle("is-visible", closed);
/* Sold overlay. SOLD_AT keeps a page reload from replaying an old sale
across the four screens. Without it, only first-seen suppression applies. */
const soldItem = control["SOLD_ITEM"] || "";
const soldPrice = control["SOLD_PRICE"] || "";
const soldAtRaw = control["SOLD_AT"] || "";
if (soldItem && soldPrice) {
const key = soldItem + "|" + soldPrice + "|" + soldAtRaw;
if (key !== lastSoldKey) {
const firstSight = lastSoldKey === null;
lastSoldKey = key;
let fresh = true;
if (soldAtRaw) {
const t = Date.parse(soldAtRaw);
if (!isNaN(t)) fresh = (Date.now() - t) < SOLD_MAX_AGE_MS;
} else if (firstSight) {
/* No SOLD_AT column yet: suppress on first load so a reload
mid-evening does not replay the last sale. */
fresh = false;
}
if (fresh) showSold(soldItem, soldPrice);
}
}
}
function showSold(name, price) {
document.getElementById("sold-name").textContent =
String(name).replace(/\*\*\*SOLD\*\*\*/gi, "").trim();
document.getElementById("sold-price").textContent = formatEuro(price);
const o = document.getElementById("overlay-sold");
o.classList.add("is-visible");
burstConfetti(SOLD_OVERLAY_MS);
setTimeout(() => o.classList.remove("is-visible"), SOLD_OVERLAY_MS);
}
/* ---------- confetti: self-contained, no CDN ---------- */
const canvas = document.getElementById("confetti-canvas");
const ctx = canvas.getContext("2d");
let confetti = [], confettiUntil = 0, confettiRunning = false;
function sizeCanvas() {
canvas.width = window.innerWidth;
canvas.height = window.innerHeight;
}
sizeCanvas();
window.addEventListener("resize", sizeCanvas);
function burstConfetti(durationMs) {
const colours = ["#f0dfa8", "#c9a227", "#ffffff", "#e8c96a"];
confetti = [];
for (let i = 0; i < 220; i++) {
confetti.push({
x: Math.random() * canvas.width,
y: -Math.random() * canvas.height,
w: 6 + Math.random() * 8,
h: 10 + Math.random() * 14,
vy: 2 + Math.random() * 4,
vx: -1 + Math.random() * 2,
rot: Math.random() * Math.PI,
vr: -0.1 + Math.random() * 0.2,
c: colours[(Math.random() * colours.length) | 0]
});
}
confettiUntil = Date.now() + durationMs;
if (!confettiRunning) { confettiRunning = true; requestAnimationFrame(tickConfetti); }
}
function tickConfetti() {
ctx.clearRect(0, 0, canvas.width, canvas.height);
if (Date.now() > confettiUntil) { confettiRunning = false; return; }
confetti.forEach(p => {
p.y += p.vy; p.x += p.vx; p.rot += p.vr;
if (p.y > canvas.height + 20) { p.y = -20; p.x = Math.random() * canvas.width; }
ctx.save();
ctx.translate(p.x, p.y);
ctx.rotate(p.rot);
ctx.fillStyle = p.c;
ctx.fillRect(-p.w / 2, -p.h / 2, p.w, p.h);
ctx.restore();
});
requestAnimationFrame(tickConfetti);
}
/* ---------- rotation ----------
Per group of 10 items:
sponsors -> items -> leaderboard pages -> bid instructions
------------------------------------------------------------------ */
function buildSequence() {
const s = [];
const lbPages = Math.max(1, Math.ceil(items.length / LB_PER_PAGE));
for (let start = 0; start < items.length; start += GROUP_SIZE) {
s.push({ type: "sponsors" });
for (let i = start; i < Math.min(start + GROUP_SIZE, items.length); i++) {
s.push({ type: "item", index: i });
}
for (let p = 0; p < lbPages; p++) {
s.push({ type: "leaderboard", page: p });
}
s.push({ type: "bid" });
}
return s;
}
function startRotation() {
seq = buildSequence();
seqIndex = 0;
runSlide();
}
function runSlide() {
clearTimeout(slideTimer);
/* Rebuild at the end of a pass so item count changes are picked up. */
if (seqIndex >= seq.length) {
seq = buildSequence();
seqIndex = 0;
}
const step = seq[seqIndex];
let duration;
if (step.type === "item") {
const item = items[step.index];
if (!item) { seqIndex++; return runSlide(); }
renderItem(item);
show("slide-item");
duration = item.video ? ITEM_MS_VIDEO : ITEM_MS_NOVIDEO;
} else if (step.type === "leaderboard") {
renderLeaderboard(step.page);
show("slide-leaderboard");
duration = LEADERBOARD_MS;
} else if (step.type === "sponsors") {
show("slide-sponsors");
duration = SPONSOR_MS;
} else {
/* Never invite bids that cannot be placed. */
const formClosed =
String(control["FORM_STATUS"] || "").trim().toUpperCase() === "CLOSED" ||
isOn(control["AUCTION_CLOSED"]);
if (formClosed) { seqIndex++; return runSlide(); }
show("slide-bid");
duration = BIDSLIDE_MS;
}
seqIndex++;
slideTimer = setTimeout(runSlide, duration);
}
function show(id) {
document.querySelectorAll(".slide").forEach(el => {
if (el.id === id) return;
el.classList.remove("is-shown");
el.classList.remove("is-active");
});
const el = document.getElementById(id);
el.classList.add("is-active");
requestAnimationFrame(() => el.classList.add("is-shown"));
}
/* ---------- renderers ---------- */
function renderItem(item) {
const frame = document.getElementById("video-frame");
const video = document.getElementById("item-video");
const wantSrc = item.video || "";
/* Only touch the video when the source changes, or the 5s refresh
would restart playback every cycle. */
if (wantSrc) {
if ((video.getAttribute("data-src") || "") !== wantSrc) {
video.setAttribute("data-src", wantSrc);
video.src = wantSrc;
video.play().catch(() => {});
}
frame.classList.add("has-video");
} else {
video.removeAttribute("data-src");
video.removeAttribute("src");
frame.classList.remove("has-video");
}
const cleanName = item.name.replace(/\*\*\*SOLD\*\*\*/gi, "").trim();
document.getElementById("video-placeholder").textContent = cleanName;
document.getElementById("title-num").textContent = item.num;
document.getElementById("title-name").textContent = item.name;
document.getElementById("item-bid").textContent = formatEuro(item.bid);
document.getElementById("bid-label").textContent =
item.sold ? "Sold For" : "Current Leading Bid";
document.getElementById("bid-col").classList.toggle("is-sold", item.sold);
const list = document.getElementById("bullets");
const lines = item.bullets
.split(/\r?\n/)
.map(l => l.replace(/^\s*\d+[.)]\s*/, "").trim())
.filter(Boolean)
.slice(0, 4);
list.innerHTML = lines.map(l => "<li>" + escapeHtml(l) + "</li>").join("");
const sponsor = document.getElementById("sponsor");
const logo = document.getElementById("sponsor-logo");
const hasSponsor = !!(item.sponsor || item.logo);
sponsor.classList.toggle("is-empty", !hasSponsor);
document.getElementById("sponsor-name").textContent = item.sponsor;
if (item.logo) {
logo.src = item.logo;
logo.classList.add("is-shown");
} else {
logo.removeAttribute("src");
logo.classList.remove("is-shown");
}
}
function renderLeaderboard(page) {
/* Sorted by item number to match the printed booklet. */
const sorted = items.slice().sort((a, b) => toNumber(a.num) - toNumber(b.num));
const pages = Math.max(1, Math.ceil(sorted.length / LB_PER_PAGE));
const p = page % pages;
const slice = sorted.slice(p * LB_PER_PAGE, (p + 1) * LB_PER_PAGE);
document.getElementById("lb-rows").innerHTML = slice.map(it =>
'<div class="lb-row">' +
'<span class="lb-num">' + escapeHtml(it.num) + '</span>' +
'<span class="lb-name">' + escapeHtml(it.name) + '</span>' +
'<span class="lb-bid">' + formatEuro(it.bid) + '</span>' +
'</div>'
).join("");
document.getElementById("lb-foot").textContent = (p + 1) + " / " + pages;
}
